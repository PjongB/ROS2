Goal pose와 결과 함수

def make_goal_pose(x, y, yaw_deg=0.0):
    q = tf_transformations.quaternion_from_euler(
        0, 0, math.radians(yaw_deg)
    )
    pose = PoseStamped()
    pose.header.frame_id = "map"
    pose.header.stamp = nav.get_clock().now().to_msg()
    pose.pose.position.x = float(x)
    pose.pose.position.y = float(y)
    pose.pose.orientation.x = q[0]
    pose.pose.orientation.y = q[1]
    pose.pose.orientation.z = q[2]
    pose.pose.orientation.w = q[3]
    return pose

def task_result_to_text(result):
    names = {
        TaskResult.SUCCEEDED: "SUCCEEDED",
        TaskResult.CANCELED: "CANCELED",
        TaskResult.FAILED: "FAILED",
    }
    return names.get(result, f"UNKNOWN: {result}")
A-4

단일 목표 주행 함수

def publish_zero_velocity():
    cmd_vel_pub.publish(Twist())

def navigate_to_pose(x, y, yaw_deg=0.0, timeout_sec=300):
    nav.goToPose(make_goal_pose(x, y, yaw_deg))
    while not nav.isTaskComplete():
        rclpy.spin_once(nav, timeout_sec=1.0)
        feedback = nav.getFeedback()
        if feedback:
            print("Distance:", round(feedback.distance_remaining, 3), "m")
            if Duration.from_msg(feedback.navigation_time) > Duration(
                seconds=timeout_sec
            ):
                nav.cancelTask()
                publish_zero_velocity()
                return {"success": False, "result": "CANCELED"}

    result = nav.getResult()
    return {
        "success": result == TaskResult.SUCCEEDED,
        "result": task_result_to_text(result),
        "target": {"x": x, "y": y, "yaw_deg": yaw_deg},
    }

    if location_name not in NAMED_LOCATIONS:
        return {
            "success": False,
            "available_locations": list(NAMED_LOCATIONS),
        }
    return navigate_to_pose(**NAMED_LOCATIONS[location_name])
A-6

제한된 cmd_vel 직접 제어

MAX_LINEAR_X = 0.12
MAX_ANGULAR_Z = 0.50
MAX_DURATION = 3.0
SPEEDS = {
    "slow": (0.05, 0.25),
    "normal": (0.10, 0.45),
}

def clamp(value, low, high):
    return max(min(float(value), high), low)

def direct_drive_motion(motion, speed, duration_sec):
    if motion not in ["forward", "backward", "turn_left", "turn_right"]:
        return {"success": False, "message": "지원하지 않는 동작"}
    if speed not in SPEEDS:
        return {"success": False, "message": "지원하지 않는 속도"}

    duration_sec = clamp(duration_sec, 0.1, MAX_DURATION)
    linear, angular = SPEEDS[speed]
    linear_x = linear if motion == "forward" else -linear if motion == "backward" else 0.0
    angular_z = angular if motion == "turn_left" else -angular if motion == "turn_right" else 0.0

    nav.cancelTask()
    try:
        for _ in range(max(1, int(duration_sec * 10))):
            msg = Twist()
            msg.linear.x = clamp(linear_x, -MAX_LINEAR_X, MAX_LINEAR_X)
            msg.angular.z = clamp(angular_z, -MAX_ANGULAR_Z, MAX_ANGULAR_Z)
            cmd_vel_pub.publish(msg)
            rclpy.spin_once(nav, timeout_sec=0.01)
            time.sleep(0.1)
    finally:
        publish_zero_velocity()
    return {"success": True, "motion": motion, "duration_sec": duration_sec}

def stop_navigation():
    nav.cancelTask()
    publish_zero_velocity()
    return {"success": True, "message": "정지 명령 전송"}

def run_navigation_function(name, args):
    if name == "navigate_to_named_location":
        return navigate_to_named_location(args["location_name"])
    if name == "navigate_to_pose":
        return navigate_to_pose(args["x"], args["y"], args["yaw_deg"])
    if name == "follow_waypoints":
        return follow_waypoints(args["points"])
    if name == "direct_drive_motion":
        return direct_drive_motion(
            args["motion"], args["speed"], args["duration_sec"]
        )
    if name == "stop_navigation":
        return stop_navigation()
    if name == "get_navigation_status":
        return {
            "success": True,
            "available_locations": list(NAMED_LOCATIONS),
        }
    return {"success": False, "message": f"알 수 없는 함수: {name}"}

자연어 Tool Calling

def ask_navigation(command):
    response = client.responses.create(
        model=MODEL,
        instructions=f"""
너는 PinkyPro 주행 명령 해석기다.
등록 위치는 {list(NAMED_LOCATIONS)}이다.
명확한 명령만 tool로 실행한다.
애매하면 확인 질문을 한다.
직접 주행은 slow를 우선하고 3초를 넘기지 않는다.
정지·취소·멈춰는 stop_navigation을 사용한다.
""",
        input=command,
        tools=tools,
        tool_choice="auto",
        parallel_tool_calls=False,
        max_output_tokens=600,
    )
    calls = [x for x in response.output if x.type == "function_call"]
    if not calls:
        return response.output_text

    outputs = []
    for call in calls:
        args = json.loads(call.arguments)
        print("[선택]", call.name, args)
        result = run_navigation_function(call.name, args)
        outputs.append({
            "type": "function_call_output",
            "call_id": call.call_id,
            "output": json.dumps(result, ensure_ascii=False),
        })

    final = client.responses.create(
        model=MODEL,
        previous_response_id=response.id,
        input=outputs,
        tools=tools,
        max_output_tokens=600,
    )
    return final.output_text

yolo_model = YOLO("yolo11n.pt")

def capture_one_frame(cam):
    frame_bgr = cam.get_frame()
    if frame_bgr is None:
        raise RuntimeError("카메라 frame 없음")
    return frame_bgr, Image.fromarray(frame_bgr[:, :, ::-1])

한 장 탐지

def detect_once_from_camera(imgsz=320, conf=0.25, show=True):
    cam = Camera()
    cam.start()
    try:
        frame_bgr, image_rgb = capture_one_frame(cam)
        result = yolo_model.predict(
            source=frame_bgr, imgsz=imgsz,
            conf=conf, verbose=False,
        )[0]
        annotated_bgr = result.plot()
        if show:
            display(image_rgb)
            display(Image.fromarray(annotated_bgr[:, :, ::-1]))
        return result, annotated_bgr
    finally:
        cam.close()
B-3

탐지 결과를 텍스트로 변환

def summarize_yolo_result(result, max_objects=10):
    if result.boxes is None or len(result.boxes) == 0:
        return {"count": 0, "summary_text": "탐지된 물체가 없습니다."}

    _, width = result.orig_shape[:2]
    items = []
    for box in result.boxes[:max_objects]:
        cls_id = int(box.cls[0])
        x1, y1, x2, y2 = [float(v) for v in box.xyxy[0]]
        center = (x1 + x2) / 2
        position = "left" if center < width * 0.4 else "right" if center > width * 0.6 else "center"
        items.append(
            f"{yolo_model.names[cls_id]} "
            f"(confidence={float(box.conf[0]):.3f}, position={position})"
        )
    return {"count": len(items), "summary_text": "\n".join(items)}

SAFE

주행 안전 기준

  • NAMED_LOCATIONS 좌표는 실제 map에서 확인한 값으로 바꿉니다.
  • 첫 시험은 바퀴를 띄우거나 넓은 공간에서 slow로 진행합니다.
  • LLM이 선택한 함수와 인자를 화면에서 확인합니다.
  • /cmd_vel은 최대 속도와 3초 timeout을 코드로 제한합니다.
  • YOLO 설명은 주행을 자동 실행하지 않으며, 사람이 안전 여부를 판단합니다.
  • 사람이나 장애물이 보이면 전진보다 정지를 우선합니다.
  • API Key를 Notebook, Git, 화면 캡처에 남기지 않습니다.
FIX

문제 해결

  • Nav2 준비에서 멈춤: terminal 2의 lifecycle, map 경로, initial pose, ROS_DOMAIN_ID를 확인합니다.
  • odom/TF를 못 읽음: PC·robot 양쪽 domain과 RMW 설정을 맞추고 실제 주행에서는 use_sim_time=false인지 확인합니다.

  • YOLO가 틀리게 설명: confidence를 높이고, LLM prompt가 YOLO 결과 외 내용을 추측하지 못하게 유지합니다.
