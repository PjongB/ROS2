def capture_one_frame(cam):
    """BGR frame과 표시용 RGB Image를 반환합니다."""
    frame_bgr = cam.get_frame()
    if frame_bgr is None:
        raise RuntimeError("카메라 frame을 가져오지 못했습니다.")
    image_rgb = Image.fromarray(frame_bgr[:, :, ::-1])
    return frame_bgr, image_rgb

def capture_one_frame_bytes(cam):
    """RGB frame을 JPEG bytes로 반환합니다."""
    _, image_rgb = capture_one_frame(cam)
    buf = io.BytesIO()
    image_rgb.save(buf, format="JPEG")
    return buf.getvalue()

완료: 출력 없이 셀이 끝납니다.

def print_detection_result(result):
    if result.boxes is None or len(result.boxes) == 0:
        print("탐지된 물체가 없습니다.")
        return

    for i, box in enumerate(result.boxes, start=1):
        cls_id = int(box.cls[0])
        name = model.names[cls_id]
        confidence = float(box.conf[0])
        x1, y1, x2, y2 = [float(v) for v in box.xyxy[0]]
        print(
            f"{i}. {name} / confidence={confidence:.2f} "
            f"/ box=({x1:.0f}, {y1:.0f}, {x2:.0f}, {y2:.0f})"
        )

print_detection_result(last_result)

짧은 실시간 탐지 함수

def run_short_realtime_detection(
    num_frames=30, imgsz=320, conf=0.25, delay=0.03,
):
    cam = Camera()
    cam.start()
    last_result = None
    last_annotated_bgr = None
    try:
        for i in range(num_frames):
            frame_bgr, _ = capture_one_frame(cam)
            last_result = model.predict(
                source=frame_bgr, imgsz=imgsz,
                conf=conf, verbose=False,
            )[0]
            last_annotated_bgr = last_result.plot()
            annotated_rgb = Image.fromarray(
                last_annotated_bgr[:, :, ::-1]
            )
            clear_output(wait=True)
            print(
                f"frame {i + 1}/{num_frames} "
                f"| imgsz={imgsz} | conf={conf}"
            )
            display(annotated_rgb)
            time.sleep(delay)
    finally:
        cam.close()
    return last_result, last_annotated_bgr

완료: 출력 없이 함수가 만들어집니다.

def compare_imgsz_once():
    cam = Camera()
    cam.start()
    try:
        frame_bgr, image_rgb = capture_one_frame(cam)
        display(image_rgb)
    finally:
        cam.close()

    rows = []
    for size in [320, 480, 640]:
        t0 = time.perf_counter()
        result = model.predict(
            source=frame_bgr, imgsz=size,
            conf=0.25, verbose=False,
        )[0]
        elapsed = time.perf_counter() - t0
        fps = 1 / elapsed if elapsed > 0 else 0
        count = 0 if result.boxes is None else len(result.boxes)
        rows.append((size, elapsed, fps, count))
        print(
            f"imgsz={size} | time={elapsed:.3f} sec "
            f"| approx FPS={fps:.2f} | detections={count}"
        )
    return rows

speed_results = compare_imgsz_once()

완료: 각 크기의 시간, FPS, 탐지 개수가 출력됩니다.

운영

성능 조절과 안전

  • imgsz 640 → 480 → 320은 대체로 빨라지지만 작은 물체를 놓칠 수 있습니다.
  • conf를 높이면 애매한 탐지는 줄지만 실제 물체도 놓칠 수 있습니다.
  • 카메라는 항상 try/finally에서 닫습니다.
  • 로봇 행동 연결 전 class allowlist, confidence threshold, 연속 frame 확인을 적용합니다.
  • 장시간 실행 전 CPU 온도와 memory를 확인하고 종료 조건을 둡니다.
해결

문제 해결

  • ModuleNotFoundError: 셀 1의 sys.executable과 package 설치 Python을 맞춥니다.
  • Camera frame 없음: cable·권한·다른 camera 사용 process를 확인합니다.
  • 색상이 이상함: BGR→RGB 변환이 한 번만 적용됐는지 확인합니다.
  • Model 다운로드 실패: Internet·DNS·저장 공간을 확인합니다.
  • 탐지가 느림: imgsz=320으로 낮추고 frame 수를 줄입니다.
  • 재실행 후 camera 오류: Kernel을 재시작하고 셀 1부터 다시 실행합니다.
